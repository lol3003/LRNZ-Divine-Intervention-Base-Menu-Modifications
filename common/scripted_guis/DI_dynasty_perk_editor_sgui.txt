### Select Character ###
DI_dynasty_selected_char = {
    scope = character

    saved_scopes = {
        target # player character
    }

    effect = {
        save_scope_as = select_char

        scope:target = {
            set_variable = {
                name = DI_dynasty_selected_char
                value = scope:select_char

            }
            set_variable = {
                name = DI_dynasty_selected_dynasty
                value = scope:select_char.dynasty
            }
        }
        debug_log = "Selected dynasty edit character set"
        # NOTE: removed debug_log = "[scope:select_char.GetFullName]" — saved scopes are not
        # resolvable in the debug_log datacontext and produced pdx_data_factory errors per click.
    }
}

### Legacy editor cheat mode toggle (Option A Test 1) ###
# Sets/clears DI_legacy_editor_mode on the player. The vanilla window override
# (gui/window_dynasty_legacy.gui) gates free perk selection on this flag AND on the viewed
# dynasty being the DI-selected one, so a stuck flag cannot affect other dynasties.
DI_dynasty_perk_editor_mode_on = {
	scope = character

	effect = {
		set_variable = {
			name = DI_legacy_editor_mode
			value = 1
		}
		debug_log = "DI legacy editor cheat mode ENABLED"
	}
}

DI_dynasty_perk_editor_mode_off = {
	scope = character

	effect = {
		remove_variable = DI_legacy_editor_mode
		debug_log = "DI legacy editor cheat mode DISABLED"
	}
}

### Select Character Init ###
DI_dynasty_perk_select_char_init = {
    scope = character

    effect = {
        if = {
            limit = {
                NOT = { has_variable = DI_dynasty_selected_char }
            }
            set_variable = {
                name = DI_dynasty_selected_char
                value = this
            }
            set_variable = {
                name = DI_dynasty_selected_dynasty
                value = this.dynasty
            }
        }
        debug_log = "Initialized dynasty edit character set"
        debug_log = "[scope:this.GetFullName]"
    }
}
